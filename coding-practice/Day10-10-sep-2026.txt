
- name: Setup Docker Containers for Multiple Apps
  hosts: all
  become: yes

  tasks:

    - name: Pull Web App Image
      community.docker.docker_image:
        name: my-web-app
        source: pull

    - name: Pull Database App Image
      community.docker.docker_image:
        name: my-database-app
        source: pull

    - name: Run Web App Container
      community.docker.docker_container:
        name: web-app
        image: my-web-app
        state: started
        published_ports:
          - "8080:80"

    - name: Run Database App Container
      community.docker.docker_container:
        name: database-app
        image: my-database-app
        state: started
        published_ports:
          - "3306:3306"


# ============================================================
# TERRAFORM — VPC WITH PUBLIC AND PRIVATE SUBNETS
# ============================================================

provider "aws" {
  region = "us-east-1"
}

resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"

  tags = {
    Name = "pathnex-vpc"
  }
}

resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "us-east-1a"
  map_public_ip_on_launch = true

  tags = {
    Name = "pathnex-public-subnet"
  }
}

resource "aws_subnet" "private" {
  vpc_id            = aws_vpc.main.id
  cidr_block        = "10.0.2.0/24"
  availability_zone = "us-east-1b"

  tags = {
    Name = "pathnex-private-subnet"
  }
}


# ============================================================
# KUBERNETES — STATEFULSET
# ============================================================

apiVersion: apps/v1
kind: StatefulSet

metadata:
  name: pathnex-db

spec:
  serviceName: "pathnex-db"
  replicas: 3

  selector:
    matchLabels:
      app: pathnex-db

  template:
    metadata:
      labels:
        app: pathnex-db

    spec:
      containers:

        - name: db
          image: mysql:8.0

          env:
            - name: MYSQL_ROOT_PASSWORD
              value: "rootpassword"

          ports:
            - containerPort: 3306

          volumeMounts:
            - mountPath: /var/lib/mysql
              name: mysql-storage

  volumeClaimTemplates:

    - metadata:
        name: mysql-storage

      spec:
        accessModes:
          - ReadWriteOnce

        resources:
          requests:
            storage: 1Gi


# ============================================================
# JENKINSFILE — BUILD, PUSH AND DEPLOY
# ============================================================

pipeline {

  agent any

  environment {
    IMAGE_NAME = 'your-dockerhub-username/pathnex-web-app'
    IMAGE_TAG = 'latest'
  }

  stages {

    stage('Build Docker Image') {

      steps {

        script {

          docker.build(
            "${IMAGE_NAME}:${IMAGE_TAG}"
          )

        }
      }
    }

    stage('Push Docker Image') {

      steps {

        script {

          docker.withRegistry(
            'https://index.docker.io/v1/',
            'docker-credentials'
          ) {

            docker.image(
              "${IMAGE_NAME}:${IMAGE_TAG}"
            ).push()

          }
        }
      }
    }

    stage('Deploy to Kubernetes') {

      steps {

        script {

          sh 'kubectl apply -f deployment.yaml'

        }
      }
    }
  }
}


# ============================================================
# GITLAB CI/CD — BUILD, PUSH AND DEPLOY
# ============================================================

stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE/pathnex-web-app"

build:

  stage: build

  script:
    - docker build -t "$IMAGE_NAME:$CI_COMMIT_SHORT_SHA" .


push:

  stage: push

  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker push "$IMAGE_NAME:$CI_COMMIT_SHORT_SHA"


deploy:

  stage: deploy

  script:
    - kubectl apply -f kubernetes/deployment.yaml
