---
- name: Deploy AWS Lambda Function
  hosts: localhost
  connection: local
  gather_facts: false

  tasks:
    - name: Create Lambda deployment package
      command: zip -r lambda.zip lambda_function.py

    - name: Create Lambda function
      command: >
        aws lambda create-function
        --function-name pathnex-lambda
        --runtime python3.12
        --role arn:aws:iam::123456789012:role/lambda-execution-role
        --handler lambda_function.lambda_handler
        --zip-file fileb://lambda.zip

#terraform 

provider "aws" {
  region = "ap-south-1"
}

resource "aws_iam_role" "lambda_exec_role" {
  name = "pathnex-lambda-execution-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"

    Statement = [
      {
        Effect = "Allow"

        Principal = {
          Service = "lambda.amazonaws.com"
        }

        Action = "sts:AssumeRole"
      }
    ]
  })
}

resource "aws_api_gateway_rest_api" "pathnex_api" {
  name        = "pathnex-api"
  description = "Pathnex Serverless API"
}

resource "aws_api_gateway_resource" "pathnex_resource" {
  rest_api_id = aws_api_gateway_rest_api.pathnex_api.id
  parent_id   = aws_api_gateway_rest_api.pathnex_api.root_resource_id
  path_part   = "pathnex"
}

resource "aws_api_gateway_method" "pathnex_method" {
  rest_api_id   = aws_api_gateway_rest_api.pathnex_api.id
  resource_id   = aws_api_gateway_resource.pathnex_resource.id
  http_method   = "GET"
  authorization = "NONE"
}

resource "aws_lambda_function" "pathnex_lambda" {
  function_name = "pathnex-lambda"
  runtime       = "python3.12"
  role          = aws_iam_role.lambda_exec_role.arn
  handler       = "lambda_function.lambda_handler"
  filename      = "lambda.zip"

  source_code_hash = filebase64sha256("lambda.zip")
}

resource "aws_api_gateway_integration" "lambda_integration" {
  rest_api_id             = aws_api_gateway_rest_api.pathnex_api.id
  resource_id             = aws_api_gateway_resource.pathnex_resource.id
  http_method             = aws_api_gateway_method.pathnex_method.http_method
  integration_http_method = "POST"
  type                    = "AWS_PROXY"

  uri = aws_lambda_function.pathnex_lambda.invoke_arn
}

resource "aws_lambda_permission" "api_gateway" {
  statement_id  = "AllowExecutionFromAPIGateway"
  action        = "lambda:InvokeFunction"
  function_name = aws_lambda_function.pathnex_lambda.function_name
  principal     = "apigateway.amazonaws.com"

  source_arn = "${aws_api_gateway_rest_api.pathnex_api.execution_arn}/*/*"
}        

#kubernetes
pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function...'

                sh '''
                    cd lambda
                    zip -r ../lambda.zip .
                '''
            }
        }

        stage('Deploy to AWS Lambda') {
            steps {
                echo 'Deploying Lambda function...'

                sh '''
                    aws lambda update-function-code \
                    --function-name pathnex-lambda \
                    --zip-file fileb://lambda.zip
                '''
            }
        }
    }
}

stages:
  - build
  - deploy

build:
  stage: build
  script:
    - cd lambda
    - zip -r ../lambda.zip .
  artifacts:
    paths:
      - lambda.zip

deploy:
  stage: deploy
  script:
    - aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip
  dependencies:
    - build
