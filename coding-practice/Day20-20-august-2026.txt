#!/usr/bin/env bash
set -euo pipefail

echo "=========================================================="
echo " Generating Day 20 Corrected Security & Hardening Suite"
echo "=========================================================="



# --------------------------------------------------
# 1. Ansible Playbook (iptables Firewall Hardening)
# --------------------------------------------------

- name: Configure Firewall with iptables
  hosts: all
  become: yes
  tasks:
    - name: Allow loopback traffic
      ansible.builtin.iptables:
        chain: INPUT
        in_interface: lo
        jump: ACCEPT

    - name: Allow established and related connections
      ansible.builtin.iptables:
        chain: INPUT
        ctstate: ESTABLISHED,RELATED
        jump: ACCEPT

    - name: Allow SSH traffic
      ansible.builtin.iptables:
        chain: INPUT
        protocol: tcp
        destination_port: "22"
        jump: ACCEPT

    - name: Allow HTTP traffic
      ansible.builtin.iptables:
        chain: INPUT
        protocol: tcp
        destination_port: "80"
        jump: ACCEPT

    - name: Set default policy for INPUT chain to DROP
      ansible.builtin.iptables:
        chain: INPUT
        policy: DROP
EOF

# --------------------------------------------------
# 2. Terraform (Security Group Hardening)
# --------------------------------------------------

resource "aws_security_group" "pathnex_sg" {
  name        = "pathnex-sg"
  description = "Security group for Pathnex web servers"
  vpc_id      = var.vpc_id

  tags = {
    Name = "pathnex-sg"
  }
}

resource "aws_security_group_rule" "allow_ssh" {
  type              = "ingress"
  from_port         = 22
  to_port           = 22
  protocol          = "tcp"
  cidr_blocks       = ["10.0.0.0/16"] # Restricted to internal network/bastion
  security_group_id = aws_security_group.pathnex_sg.id
}

resource "aws_security_group_rule" "allow_http" {
  type              = "ingress"
  from_port         = 80
  to_port           = 80
  protocol          = "tcp"
  cidr_blocks       = ["0.0.0.0/0"]
  security_group_id = aws_security_group.pathnex_sg.id
}

resource "aws_security_group_rule" "allow_egress_all" {
  type              = "egress"
  from_port         = 0
  to_port           = 0
  protocol          = "-1"
  cidr_blocks       = ["0.0.0.0/0"]
  security_group_id = aws_security_group.pathnex_sg.id
}
EOF

# --------------------------------------------------
# 3. Kubernetes (Least-Privilege RBAC)
# --------------------------------------------------

apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  namespace: production
  name: pathnex-app-developer
rules:
  - apiGroups: ["", "apps"]
    resources: ["pods", "deployments", "services"]
    verbs: ["get", "list", "watch", "create", "update"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: pathnex-developer-binding
  namespace: production
subjects:
  - kind: User
    name: pathnex-user
    apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: Role
  name: pathnex-app-developer
  apiGroup: rbac.authorization.k8s.io
EOF

# --------------------------------------------------
# 4. Jenkinsfile (Secure Credentials Management)
# --------------------------------------------------

pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('pathnex-web-app')
                }
            }
        }
        stage('Push') {
            steps {
                script {
                    docker.withRegistry('https://index.docker.io/v1/', 'docker-credentials') {
                        def app = docker.image('pathnex-web-app')
                        app.push('latest')
                    }
                }
            }
        }
    }
}
EOF

# --------------------------------------------------
# 5. GitLab CI/CD (Secure Pipeline Deployment)
# --------------------------------------------------

stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD" $CI_REGISTRY
    - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
    - kubectl rollout status deployment/pathnex-web-app --timeout=60s
