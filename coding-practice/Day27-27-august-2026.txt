---
- name: Deploy serverless lambda function
hosts: localhost
connection:local
gather_facts: false

tasks:
  - name: Create Lambda function
  command:
    argv: 
    - aws
    - lambda
    - create-function
    - --function-name
    - pathnex-lambda
    - --runtime
    - nodejs20.x
    - --role
    - arn:aws:iam::123456789012:role/Lambda
    - --handler
    - index.handler
    - --zip-file
    - fil



  resource "aws_dynamodb_table" "pathnex_table" {
  name         = "pathnex-table"
  billing_mode = "PAY_PER_REQUEST"
  hash_key     = "id"

  attribute {
    name = "id"
    type = "S"
  }

  tags = {
    Name = "pathnex-table"
  }
}

apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
  name: pathnex-lambda
  namespace: default

spec:
  runtime: nodejs14
  handler: index.handler
  function: |
    module.exports = function(event, context) {
      context.succeed('Hello World');
    };


pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function package...'
                sh 'cd lambda && zip -r ../lambda.zip .'
            }
        }

        stage('Deploy to Lambda') {
            steps {
                echo 'Deploying Lambda function...'
                sh '''
                    aws lambda update-function-code \
                    --function-name pathnex-lambda \
                    --zip-file fileb://lambda.zip
                '''
            }
        }
    }
}
