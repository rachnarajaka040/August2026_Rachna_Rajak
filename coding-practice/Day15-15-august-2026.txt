   ## 2. Terraform — Create EC2 with ALB      

   provider "aws" {
  region = "ap-south-1"
}

resource "aws_security_group" "allow_ssh_http" {
  name        = "pathnex-web-sg"
  description = "Allow SSH and HTTP traffic"
  vpc_id      = aws_vpc.main.id

  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    description = "SSH"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["YOUR_IP/32"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "pathnex_ec2" {
  ami                    = "YOUR_VALID_AMI_ID"
  instance_type          = "t2.micro"
  vpc_security_group_ids = [aws_security_group.allow_ssh_http.id]
  subnet_id              = aws_subnet.public_1.id

  tags = {
    Name = "Pathnex-EC2"
  }
}

resource "aws_lb" "pathnex_lb" {
  name               = "pathnex-lb"
  internal           = false
  load_balancer_type = "application"

  security_groups = [
    aws_security_group.allow_ssh_http.id
  ]

  subnets = [
    aws_subnet.public_1.id,
    aws_subnet.public_2.id
  ]

  tags = {
    Name = "Pathnex-ALB"
  }
}

resource "aws_lb_target_group" "pathnex_target_group" {
  name     = "pathnex-target-group"
  port     = 80
  protocol = "HTTP"
  vpc_id   = aws_vpc.main.id

  health_check {
    path = "/"
  }
}

resource "aws_lb_target_group_attachment" "pathnex_target" {
  target_group_arn = aws_lb_target_group.pathnex_target_group.arn
  target_id        = aws_instance.pathnex_ec2.id
  port             = 80
}

resource "aws_lb_listener" "http" {
  load_balancer_arn = aws_lb.pathnex_lb.arn
  port              = 80
  protocol          = "HTTP"

  default_action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.pathnex_target_group.arn
  }
}

## 3. Jenkinsfile — Multi-Environment Deployment
pipeline {
    agent any

    environment {
        IMAGE_NAME = "pathnex-web-app"
        DEPLOY_ENV = "production"
    }

    stages {

        stage("Build") {
            steps {
                echo "Building Docker image..."

                script {
                    docker.build("${IMAGE_NAME}:${BUILD_NUMBER}")
                }
            }
        }

        stage("Test") {
            steps {
                echo "Running tests..."

                sh "echo Running application tests..."
            }
        }

        stage("Deploy to Kubernetes") {
            steps {
                script {

                    if (env.DEPLOY_ENV == "production") {

                        echo "Deploying to production..."

                        sh "kubectl apply -f kubernetes/prod-deployment.yaml"

                    } else {

                        echo "Deploying to development..."

                        sh "kubectl apply -f kubernetes/dev-deployment.yaml"
                    }
                }
            }
        }
    }

    post {
        success {
            echo "Pipeline completed successfully!"
        }

        failure {
            echo "Pipeline failed!"
        }
    }
}
