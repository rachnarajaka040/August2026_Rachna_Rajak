#Ansible — Create EC2 Instances 
- name: Create Ec2 instances
 hosts: localhost
 become: no
 collections:
  - amazon.aws
tasks:
   - name: Launch Ec2 instance in us-east-1  
   amazon.aws.ec2_instance:
   name: pathnex-ec2-us-east-1
   key_name: pathnex-key
   region: us-east-1
   instance_type: t2.micro
   image_id: ami-0abcd1234abcd1234
   wait:true

#Terraform — Multi-Region VPC with EC2

provider "aws"{
  alias = "us_east"
  region = "us-east-1"
}

provider "aws"
{
  alias = "us_west"
  region = "us-west-1"
}

resource "aws_vpc" "us_east_vpc"
{
  provider = aws.us_east
  cidr_block = "10.0.0.0/16"
}

resource "aws_vpc" "us_west_vpc{
  provider = aws.us_west
  cidr_block ="10.1.0.0/16"
}

#jenkins deploy
pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                script {
                    echo 'Building Docker image...'
                    docker.build('pathnex-web-app')
                }
            }
        }

        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.CLUSTER == 'us-east') {
                        sh 'kubectl --context=us-east apply -f kubernetes/deployment.yaml'
                    } else if (env.CLUSTER == 'us-west') {
                        sh 'kubectl --context=us-west apply -f kubernetes/deployment.yaml'
                    } else {
                        error 'Invalid CLUSTER value. Use us-east or us-west.'
                    }
                }
            }
        }
    }
}
