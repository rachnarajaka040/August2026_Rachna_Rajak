
Ansible — Install Docker

- name: Install Docker
  hosts: all
  become: yes
  tasks:
    - name: Install Docker
      yum:
        name: docker
        state: present
    - name: Start Docker
      service:
        name: docker
        state: started
        enabled: yes

Terraform — Provisioning EC2 with VPC

resource "aws_vpc" "main" {

Kubernetes — Pod with Volume Mount

apiVersion: v1
kind: Pod
metadata:
  name: rachna-pod
spec:
  containers:
    - name: web
      image: nginx
      volumeMounts:
        - mountPath: /usr/share/nginx/html
          name: html-volume
  volumes:
    - name: html-volume
      hostPath:
        path: /data
        type: Directory

Shell Script — Check for Running Services

#!/bin/bash
services=("nginx" "docker" "httpd")
for service in "${services[@]}"
do
  if systemctl is-active --quiet $service; then
    echo "$service is running"
  else
    echo "$service is not running"
  fi
done
