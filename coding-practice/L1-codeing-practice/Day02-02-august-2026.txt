#Ansible Task — Install Nginx on Pathnex Server

- name: install nginx server
  hosts: all
  become: true
  tasks:
      - name: install nginx
        yum:
          name: nginx
          state: present

      - name: enable nginx
        service:
         name:  nginx
         state: started
         enable: yes
